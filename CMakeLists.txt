cmake_minimum_required(VERSION 3.20)
project(CoreLens VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(corelens_lib
    src/json.cpp src/process.cpp src/perf.cpp src/assembly.cpp)
target_include_directories(corelens_lib PUBLIC include)
find_package(Threads REQUIRED)
target_link_libraries(corelens_lib PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(corelens_lib PRIVATE /W4)
else()
    target_compile_options(corelens_lib PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(corelens-native src/main.cpp)
target_link_libraries(corelens-native PRIVATE corelens_lib)
include(GNUInstallDirs)
install(TARGETS corelens-native RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
option(CORELENS_BUILD_TESTS "Build native parser and process tests" OFF)
if(CORELENS_BUILD_TESTS)
    enable_testing()
    add_executable(corelens-tests tests/native_tests.cpp)
    target_link_libraries(corelens-tests PRIVATE corelens_lib)
    add_test(NAME corelens-native-tests COMMAND corelens-tests)
endif()
